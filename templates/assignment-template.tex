\begin{problem}
\verb|codingbox| defaults to Python, but any language \texttt{listings} ships a
\verb|\lstdefinelanguage| for works via its \texttt{language} option, e.g.
\verb|[language=C++]| or \verb|[language=bash]|.
\begin{codingbox}[language=C++]
#include <iostream>

int main() {
    // Compute the first 10 Fibonacci numbers
    int a = 0, b = 1;
    for (int i = 0; i < 10; ++i) {
        std::cout << a << " ";
        int next = a + b;
        a = b;
        b = next;
    }
    return 0;
}
\end{codingbox}
\begin{codingbox}[language=bash]
#!/usr/bin/env bash
# Print the first 10 Fibonacci numbers
a=0
b=1
for _ in $(seq 1 10); do
    echo -n "$a "
    next=$((a + b))
    a=$b
    b=$next
done
\end{codingbox}
\end{problem}
